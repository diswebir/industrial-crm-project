#  App private-path pointer (split deployment on cPanel)
# ------------------------------------------------------------------
#  In a "split" deployment, this file (renamed to app-path.txt) sits next
#  to index.php inside the web root and contains the ABSOLUTE path of the
#  private application folder, for example:
#
#      /home/youruser/private_pro/project1/myc
#

#  parent of public/ (standard single-folder layout).
#

#  access by .htaccess.
